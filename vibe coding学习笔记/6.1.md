Models
终端准备：
pip install -U "langchain[deepseek]"
初始化聊天模型：
import os
#导入python自带的os模块，用于配置环境变量和文件路径。这段代码中用于设置API Key

from langchain.chat_models import init_chat_model
#init_chat_model：初始化聊天模型的函数

os.environ['DeepSeek_API_KEY'] = 'sk-5944a2ff00f841bd9814f71521788440'
#os.environ['变量名'] = '变量值'

model = init_chat_model('deepseek-v4-flash')
#调用init_chat_model函数，变量是模型名称，将函数返回的结果保存到model变量中

访问模型：
1. 阻塞式访问：invoke
2. 流式访问：stream
response = model.invoke('你是谁？')
#模型接受两种信息的输入。
#第一种是字符串。LangChain 会默认把它理解成用户消息

response = model.invoke([
	{'role':'system','content':"你扮演火箭队中的武藏，以她的性格口吻回答用户的问题"},
	{'role':'user','content':"你是谁？"}
])
#第二种是消息聊天列表。

'''
response = model.invoke(
{'messages':[{'role':'user','content':"你是谁？"}]}
)
'''
#这里报错了。Agent 的输入格式 和 普通模型 model 的输入格式不同

print(response.content)
#这样只打印模型回复的正文，而不是整个 AIMessage 对象
